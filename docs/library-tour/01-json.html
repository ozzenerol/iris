    Library Tour::Text and data

    JSON

    The json module turns a struct marked @json into JSON text and reads it back. The compiler writes the code for each struct, so nothing is looked up by name at run time.

    Encoding

    json.encode(v) gives the JSON text of a value as a new String:

    import json

enum Proto { http, grpc }

// A service, as a service registry lists it.
@json
struct Service {
    i64 id
    String name
    Proto proto
    bool healthy = true
    str[] addrs
}

void main() {
    str[2] addrs = ["10.0.0.11:8080", "10.0.0.12:8080"]
    s := Service{id: 7, name: "billing".copy(), proto: .grpc, addrs: addrs}

    scoped String text = json.encode(s)
    print(text)

    s.name.free()
}
    {"id":7,"name":"billing","proto":"grpc","healthy":true,"addrs":["10.0.0.11:8080","10.0.0.12:8080"]}

    Arrays, slices and plain values encode too. json.encode(services.items()) gives a JSON array. An enum is written as the name of its value.

    Decoding

    json.decode(T, text) reads a T from JSON text and returns a !T. Keys the struct doesn't have are skipped. Fields whose keys are missing keep their defaults.

    import json

enum Proto { http, grpc }

@json
struct Check {
    String path
    u16 port
}

@json
struct Service {
    i64 id
    String name
    Proto proto = .http
    bool healthy = true
    String[] addrs
    Check check
    f64 load
}

!void main() {
    text := r"""
        {"id": 7, "name": "billing", "addrs": ["10.0.0.11:8080", "10.0.0.12:8080"],
         "check": {"path": "/health", "port": 9090}, "load": 0.25, "region": "eu"}
        """

    scoped Service s = try json.decode(Service, text)
    print("{s.id} {s.name} {s.proto} {s.healthy} {s.addrs.len} {s.check.path} {s.check.port} {s.load}")
}
    7 billing http true 2 /health 9090 0.25

    null leaves a field as it is. A decoded struct owns its Strings and slices. @json gives the struct a free method that frees them, so scoped works on it. When decode fails, it has already freed what it read.

    A JSON array is read into a slice, written as the type: json.decode(Service[], text). The slice is new memory. The caller frees each item that owns memory, then the slice:

    import json

@json
struct Backend {
    String addr
    i64 weight
}

!void main() {
    backends := try json.decode(Backend[], r"""[{"addr": "10.0.0.11:8080", "weight": 3}, {"addr": "10.0.0.12:8080", "weight": 1}]""")
    defer free(backends)
    defer {
        for b in backends {
            b.free()
        }
    }

    for b in backends {
        print("{b.addr} weight {b.weight}")
    }

    ports := try json.decode(u16[], "[80, 443, 8080]")
    print("{ports.len} ports, the last {ports[2]}")
    free(ports)
}
    10.0.0.11:8080 weight 3
10.0.0.12:8080 weight 1
3 ports, the last 8080

    Lists

    A Duration field reads a number of nanoseconds or text like "30s", "250ms" or "1m30s", so a configuration file can say timeout: 5s. encode writes it as nanoseconds.

    A List field is a JSON array too. decode fills the list, and push keeps growing it afterwards. The struct's free frees the list and its items. json.decode(List[T], text) reads a whole array into a new list, which the caller frees.

    import json

@json
struct Backend {
    String addr
    i64 weight = 1
}

// The upstreams of a reverse proxy.
@json
struct Pool {
    String name
    List[Backend] backends
}

!void main() {
    scoped Pool pool = try json.decode(Pool, r"""
        {"name": "billing", "backends": [{"addr": "10.0.0.11:8080", "weight": 3}, {"addr": "10.0.0.12:8080"}]}
        """)
    pool.backends.push(Backend{addr: "10.0.0.13:8080".copy()})

    for b in pool.backends {
        print("{b.addr} weight {b.weight}")
    }

    scoped String text = json.encode(pool)
    print(text)

    ports := try json.decode(List[u16], "[80, 443]")
    ports.push(8080)
    print("{ports.len} ports, the last {ports.last()}")
    ports.free()
}
    10.0.0.11:8080 weight 3
10.0.0.12:8080 weight 1
10.0.0.13:8080 weight 1
{"name":"billing","backends":[{"addr":"10.0.0.11:8080","weight":3},{"addr":"10.0.0.12:8080","weight":1},{"addr":"10.0.0.13:8080","weight":1}]}
3 ports, the last 8080

    The second backend has no "weight", so it keeps the default, 1.

    Maps

    A Map with str or String keys is a JSON object whose keys aren't known in advance, like the headers a route adds. decode fills the map, and a key given twice keeps its last value. encode writes the keys in the map's own order, which isn't the order they were added in. The struct's free frees the map, its keys and its values. json.decode(Map[str, T], text) reads a whole object into a new map.

    import json

@json
struct Route {
    String prefix
    Map[str, String] headers
}

!void main() {
    scoped Route r = try json.decode(Route, r"""
        {"prefix": "/api", "headers": {"x-team": "billing", "cache-control": "no-store"}}
        """)
    team, _ := r.headers.get("x-team")
    print("{r.headers.len} headers, x-team is {team}")

    limits := try json.decode(Map[str, i64], r"""{"api": 100, "web": 20}""")
    print("api {limits.getOr("api", 0)}, web {limits.getOr("web", 0)}")
    limits.free()
}
    2 headers, x-team is billing
api 100, web 20

    Enums

    An enum field is a JSON string that names one of the enum's values. decode turns the string back into the value. A slice of an enum is read from an array of strings. A string that names no value is a wrongType error.

    import json

enum Level { debug, info, warn, error }
enum Proto { http, grpc }

// One rule of an alerting config.
@json
struct Rule {
    String name
    Level level = .info
    Proto[] protos
}

void main() {
    text := r"""
        [{"name": "slow requests", "level": "warn", "protos": ["http", "grpc"]},
         {"name": "crashes", "level": "error", "protos": ["grpc"]},
         {"name": "deploys", "protos": []}]
        """

    rules := json.decode(Rule[], text) catch e {
        print("{e.code}: {e.message}")
        return
    }
    defer free(rules)
    defer {
        for r in rules {
            r.free()
        }
    }

    for r in rules {
        action := match r.level {
            .error => "page someone"
            .warn => "post in the channel"
            _ => "only log it"
        }
        print("{r.name}: {r.level}, {r.protos.len} protocols, {action}")
    }

    scoped String back = json.encode(rules[0])
    print(back)

    _ = json.decode(Rule, r"""{"name": "x", "level": "fatal"}""") catch e {
        print("{e.code}: {e.message}")
        Rule{}
    }
}
    slow requests: warn, 2 protocols, post in the channel
crashes: error, 1 protocols, page someone
deploys: info, 0 protocols, only log it
{"name":"slow requests","level":"warn","protos":["http","grpc"]}
JsonError.wrongType: level: "fatal" isn't one of the enum's values

    The third rule has no "level", so it keeps the default, .info.

    Keys and required fields

    A field's key is its name as written. @json("key") gives it another key. @json(required) makes decode fail when the key is missing. The two combine as @json("id", required).

    import json

// One line of a JSON access log.
@json
struct Access {
    @json("request_id", required) i64 id
    @json("remote_addr") String remote
    i64 status
}

void main() {
    scoped Access a = json.decode(Access, r"""{"request_id": 3, "remote_addr": "10.0.0.5", "status": 200}""") catch err {
        print(err.message)
        Access{}
    }
    print("{a.id} {a.remote} {a.status}")

    scoped String back = json.encode(a)
    print(back)

    _ = json.decode(Access, r"""{"status": 502}""") catch err {
        print("{err.code}: {err.message}")
        Access{}
    }
}
    3 10.0.0.5 200
{"request_id":3,"remote_addr":"10.0.0.5","status":200}
JsonError.missing: the required field "request_id" is missing

    What a @json struct can hold

    • integers, floats and bool
    • String and plain enums
    • other @json structs
    • slices, arrays and Lists of these
    • Maps from str or String keys to these

    The fields of an embedded struct appear as fields of the outer one. A str field encodes but doesn't decode, because a str can't own the text it points to. Decoded text goes in a String.

    Errors

    The errors are values of json.JsonError. The message says where in the text the problem is.

    CodeWhen
    syntaxthe text isn't JSON
    wrongTypea value has the wrong type for its field
    missinga required key isn't there
    rangea number doesn't fit its field
    import json

@json
struct Listen {
    i64 backlog
    u8 ttl
    String[] hosts
}

void main() {
    for text in [r"""{"backlog": "many"}""", r"""{"ttl": 300}""", r"""{"hosts": [1]}""", r"{oops"] {
        _ = json.decode(Listen, text) catch e {
            print("{e.code}: {e.message}")
            Listen{}
        }
    }
}
    JsonError.wrongType: backlog: expected a number, found a string
JsonError.range: ttl: 300 doesn't fit in u8
JsonError.wrongType: hosts[0]: expected a string, found a number
JsonError.syntax: line 1, column 2: expected a key in quotes

    Quoting text

    json.quote(text) gives text as a JSON string in quotes, with the characters that need it escaped. Use it to write a bit of JSON by hand:

    import json

void main() {
    msg := "upstream \"billing\" timed out\nafter 30s"
    scoped String q = json.quote(msg)
    print("{{\"error\": {q}}}")
}
    {"error": "upstream \"billing\" timed out\nafter 30s"}
